182void ESPGPS::calculateChecksum(void)
183{
184 int16_t sum = 0;
185 int16_t inx = 0;
186 char tmp;
187
188 // The checksum calc starts after '$' and ends before '*'
189 for (inx = 1; inx < 200; inx++)
190 {
191 tmp = espData->gps.nmea[inx];
192
193 // * Indicates end of data and start of checksum
194 if (tmp == '*')
195 {
196 break;
197 }
198
199 sum ^= tmp; // Build checksum
200 }
201
202 byte chk = (sum >> 4);
203 char hex[2] = { espData->gps.asciiHex[chk], 0 };
204 strcat(espData->gps.nmea, hex);
205
206 chk = (sum % 16);
207 char hex2[2] = { espData->gps.asciiHex[chk], 0 };
208 strcat(espData->gps.nmea, hex2);
209}
