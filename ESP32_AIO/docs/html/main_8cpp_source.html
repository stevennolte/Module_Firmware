287 Update.printError(Serial);

290

292 if (Update.write(data, len) != len) {

295
296 // If the upload is complete
297 if (final) {
298 if (Update.end(true)) { // True to set the size correctly
299 Serial.printf("Update Success: %u bytes\n", index + len);
