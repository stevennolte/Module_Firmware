87// Detect current system error state based on ESPdata
88LEDState MyLED::detectErrorState() {
89 int errorCount = 0;
90 LEDState primaryError = LEDState::NO_ERROR;
91
92 // Check configuration load result
93 if (espData->program.confRes != 1) {
94 errorCount++;
95 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::CONFIG_ERROR;
96 }
97
98 // Check MCP23017 state
99 if (espData->program.mcpState != 1) {
100 errorCount++;
101 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::MCP_ERROR;
102 }
103
104 // Check ADS1115 state
105 if (espData->program.adsState != 1) {
106 errorCount++;
107 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::ADS_ERROR;
108 }
109
110 // Check I2C communication state
111 if (espData->program.twoWireState != 1) {
112 errorCount++;
113 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::I2C_ERROR;
114 }
115
116 // Check GPS/IMU state
117 if (espData->gps.imuState != 1) {
118 errorCount++;
119 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::GPS_ERROR;
120 }
121
122 // Check WiFi state (only if WiFi is expected to be connected)
123 if (espData->wifi.state == 0) {
124 errorCount++;
125 if (primaryError == LEDState::NO_ERROR) primaryError = LEDState::WIFI_ERROR;
126 }
127
128 // Check if in recovery mode (program state != 1) - this indicates system-level issue
129 if (espData->program.state == 0 || espData->program.confRes == 0) {
130 return LEDState::RECOVERY_MODE;
131 }
132
133 // Return multiple errors if more than one issue detected
134 if (errorCount > 1) {
135 return LEDState::MULTIPLE_ERRORS;
136 }
137
138 return primaryError;
139}

141// Update LED display based on current error state
142void MyLED::updateErrorDisplay() {
143 switch (currentErrorState) {
144 case LEDState::NO_ERROR:
145 // Always show green when no errors are detected
146 pixel.setPixelColor(0, pixel.Color(0, 255, 0)); // Green - All systems normal
147 break;
148
149 case LEDState::SPECIAL_MODE:
150 // Rainbow effect for special mode
151 pixel.setPixelColor(0, pixel.gamma32(pixel.ColorHSV(firstPixelHue)));
152 firstPixelHue = firstPixelHue + 256;
153 if (firstPixelHue > 5*65536) {
154 firstPixelHue = 0;
155 }
156 break;
157
158 case LEDState::MULTIPLE_ERRORS:
159 case LEDState::RECOVERY_MODE:
160 handleBlinkingStates(currentErrorState);
161 return; // Skip pixel.show() as it's handled in handleBlinkingStates
162
163 default:
164 // Solid color for single error states
165 pixel.setPixelColor(0, getErrorColor(currentErrorState));
166 break;
167 }
168
169 pixel.show();
170}

201void MyLED::handleBlinkingStates(LEDState state) {
202 unsigned long currentTime = millis();
203 unsigned long blinkInterval;
204
205 if (state == LEDState::MULTIPLE_ERRORS) {
206 blinkInterval = 200; // Fast blink for multiple errors (5 Hz)

210 blinkInterval = 300; // Default blink rate for other blinking states
211 }
212
