464void MCPManager::updateLED(LEDIndicator& led) {
465 unsigned long currentTime = millis();
466 bool shouldBeOn = false;
467
468 switch (led.state) {
469 case MCPLEDState::OFF:
470 shouldBeOn = false;
471 break;
472
473 case MCPLEDState::ON:
474 shouldBeOn = true;
475 break;
476
477 case MCPLEDState::SLOW_PULSE: // 0.5 Hz (1 second on, 1 second off)
478 if (currentTime - led.lastToggle >= 1000) {
479 led.currentState = !led.currentState;
480 led.lastToggle = currentTime;
481 }
482 shouldBeOn = led.currentState;
483 break;
484
485 case MCPLEDState::FAST_PULSE: // 2 Hz (250ms on, 250ms off)
486 if (currentTime - led.lastToggle >= 250) {
487 led.currentState = !led.currentState;
488 led.lastToggle = currentTime;
489 }
490 shouldBeOn = led.currentState;
491 break;
492
493 case MCPLEDState::RAPID_PULSE: // 5 Hz (100ms on, 100ms off)
494 if (currentTime - led.lastToggle >= 100) {
495 led.currentState = !led.currentState;
496 led.lastToggle = currentTime;
497 }
498 shouldBeOn = led.currentState;
499 break;
500
501 case MCPLEDState::ERROR_FLASH: // 3 quick flashes, then pause
502 if (led.flashCount < 6) { // 3 on/off cycles = 6 state changes
503 if (currentTime - led.lastToggle >= 100) {
504 led.currentState = !led.currentState;
505 led.lastToggle = currentTime;
506 led.flashCount++;
507 }
508 shouldBeOn = led.currentState;
509 } else {
510 // Pause between flash sequences
511 if (currentTime - led.lastToggle >= 1000) {
512 led.flashCount = 0;
513 led.currentState = false;
514 led.lastToggle = currentTime;
515 }
516 shouldBeOn = false;
517 }
518 break;
519
520 case MCPLEDState::HEARTBEAT: // Double pulse pattern
521 {
522 unsigned long cycleTime = currentTime % 2000; // 2 second cycle
523 if ((cycleTime < 100) || (cycleTime >= 200 && cycleTime < 300)) {
524 shouldBeOn = true;
525 } else {
526 shouldBeOn = false;
527 }
528 }
529 break;
530 }
531
532 // Update the physical LED only if state changed
533 if (shouldBeOn != (mcp.digitalRead(led.pin) == HIGH)) {
534 mcp.digitalWrite(led.pin, shouldBeOn ? HIGH : LOW);
535 }
536}
