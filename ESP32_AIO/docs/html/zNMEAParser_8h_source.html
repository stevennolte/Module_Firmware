226 /*
227 convert a one hex digit char into an int. Used for the CRC
228 */
229 static int8_t hexToNum(const char inChar)
230 {
231 if (isdigit(inChar)) return inChar - '0';
232 else if (isupper(inChar) && inChar <= 'F') return inChar - 'A' + 10;
233 else if (islower(inChar) && inChar <= 'f') return inChar - 'a' + 10;
234 else return -1;
235 }

237 static bool strnwcmp(const char *s1, const char *s2, uint8_t len)
238 {
239 while (len-- > 0) {
240 if (*s1 != *s2 && *s1 != '-' && *s2 != '-') return false;
241 s1++;
242 s2++;
243 }
244 return true;
245 }

382 /*
383 Give a character to the parser
384 */
385 void operator<<(char inChar)
386 {
387 int8_t tmp;
388
389 switch (mState) {
390
391 /* Waiting for the starting $ character */
392 case INIT:
393 mError = NMEA::NO_ERROR;
394 if (inChar == '$') {
395 mComputedCRC = 0;
396 mState = SENT;
397 }
398 else unexpectedChar();
399 break;
400
401 case SENT:
402 if (isalnum(inChar)) {
403 if (spaceAvail()) {
404 if (mIndex < 5) {
405 mBuffer[mIndex++] = inChar;
406 mComputedCRC ^= inChar;
407 }
408 else {
409 typeTooLong();
410 }
411 }
412 else bufferFull();
413 }
414 else {
415 switch (inChar) {
416 case ',' :
417 mComputedCRC ^= inChar;
418 mBuffer[--mArgIndex] = mIndex;
419 mState = ARG;
420 break;
421 case '*' :
422 mGotCRC = 0;
423 mBuffer[--mArgIndex] = mIndex;
424 mState = CRCH;
425 break;
426 default :
427 unexpectedChar();
428 break;
429 }
430 }
431 break;
432
433 case ARG:
434 if (spaceAvail()) {
435 switch (inChar) {
436 case ',' :
437 mComputedCRC ^= inChar;
438 mBuffer[--mArgIndex] = mIndex;
439 break;
440 case '*' :
441 mGotCRC = 0;
442 mBuffer[--mArgIndex] = mIndex;
443 mState = CRCH;
444 break;
445 default :
446 mComputedCRC ^= inChar;
447 mBuffer[mIndex++] = inChar;
448 break;
449 }
450 }
451 else bufferFull();
452 break;
453
454 case CRCH:
455 tmp = hexToNum(inChar);
456 if (tmp != -1) {
457 mGotCRC |= (uint8_t)tmp << 4;
458 mState = CRCL;
459 }
460 else unexpectedChar();
461 break;
462
463 case CRCL:
464 tmp = hexToNum(inChar);
465 if (tmp != -1) {
466 mGotCRC |= (uint8_t)tmp;
467 mState = CRLFCR;
468 }
469 else unexpectedChar();
470 break;
471
472 case CRLFCR:
473 if (inChar == '\r') {
474 mState = CRLFLF;
475 }
476 else unexpectedChar();
477 break;
478
479 case CRLFLF:
480 if (inChar == '\n') {
481 if (mHandleCRC && (mGotCRC != mComputedCRC)) {
482 crcError();
483 }
484 else {
485 processSentence();
486 }
487 reset();
488 }
489 else unexpectedChar();
490 break;
491
492 default:
493 internalError();
494 break;
495 }
496 }
