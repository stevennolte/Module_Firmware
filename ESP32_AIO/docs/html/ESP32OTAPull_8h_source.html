51 int DownloadJson(const char* URL, String& payload)
52 {
53 HTTPClient http;
54 http.begin(URL);
55
56 // Send HTTP GET request
57 int httpResponseCode = http.GET();
58
59 if (httpResponseCode == 200)
60 {
61 payload = http.getString();
62 }
63

69 int DoOTAUpdate(const char* URL, ActionType Action)
70 {
71 HTTPClient http;
72 http.begin(URL);
73
74 // Send HTTP GET request
75 int httpResponseCode = http.GET();
76
77 if (httpResponseCode == 200)
78 {
79 int totalLength = http.getSize();
80
81 // this is required to start firmware update process
82 if (!Update.begin(UPDATE_SIZE_UNKNOWN))
83 return OTA_UPDATE_FAIL;
84
85 // create buffer for read
86 uint8_t buff[1280] = { 0 };
87
88 // get tcp stream
89 WiFiClient* stream = http.getStreamPtr();
90
91 // read all data from server
92 int offset = 0;
93 while (http.connected() && offset < totalLength)
94 {
95 size_t sizeAvail = stream->available();
96 if (sizeAvail > 0)
97 {
98 size_t bytes_to_read = min(sizeAvail, sizeof(buff));
99 size_t bytes_read = stream->readBytes(buff, bytes_to_read);
100 size_t bytes_written = Update.write(buff, bytes_read);
101 if (bytes_read != bytes_written)
102 {
103 // Serial.printf("Unexpected error in OTA: %d %d %d\n", bytes_to_read, bytes_read, bytes_written);
104 break;
105 }
106 offset += bytes_written;
107 if (Callback != NULL)
108 Callback(offset, totalLength);
109 }
110 }
111
112 if (offset == totalLength)
113 {
114 Update.end(true);
115 delay(1000);
116
117 // Restart ESP32 to see changes
118 if (Action == UPDATE_BUT_NO_BOOT)
119 return UPDATE_OK;
120 ESP.restart();
121 }
122 return WRITE_ERROR;
123 }
124
125 http.end();
126 return httpResponseCode;
127 }

187 int CheckForOTAUpdate(const char* server_address ,const char* JSON_PATH, const char *CurrentVersion, ActionType Action = UPDATE_AND_BOOT)
188 {
189 CurrentVersion = CurrentVersion == NULL ? "" : CurrentVersion;
190 char JSON_URL[300];
191 char firmwareURL[300];
192 strcpy(firmwareURL, server_address);
193 strcpy(JSON_URL, server_address);
194 strcat(JSON_URL, JSON_PATH);
195 // sprintf(JSON_URL,server_address,JSON_PATH);
196 // Downloading OTA Json...
197 String Payload;
198 Serial.println(JSON_URL);
199 int httpResponseCode = DownloadJson(JSON_URL, Payload);
200 if (httpResponseCode != 200)
201
202 return httpResponseCode > 0 ? httpResponseCode : HTTP_FAILED;
203
204 // Deserialize the JSON file downloaded from user's site
205 DynamicJsonDocument doc(6000);
206 DeserializationError deserialization = deserializeJson(doc, Payload.c_str());
207 if (deserialization != DeserializationError::Ok)
208 return JSON_PROBLEM;
209
210 String DeviceName = Device.isEmpty() ? (String)(WiFi.localIP()[3]) : Device;
211 String BoardName = Board.isEmpty() ? ARDUINO_BOARD : Board;
212 String ConfigName = Config.isEmpty() ? "" : Config;
213 bool foundProfile = false;
214
215 // Step through the configurations looking for a match
216 for (auto config : doc["Configurations"].as<JsonArray>())
217 {
218 String CBoard = config["Board"].isNull() ? "" : (const char *)config["Board"];
219 String CDevice = config["Device"].isNull() ? "" : (const char *)config["Device"];
220 CVersion = config["Version"].isNull() ? "" : (const char *)config["Version"];
221 strcat(firmwareURL, config["URL"]);
222 String CConfig = config["Config"].isNull() ? "" : (const char *)config["Config"];
223 if ((CBoard.isEmpty() || CBoard == BoardName) &&
224 (CDevice.isEmpty() || CDevice == DeviceName) &&
225 (CConfig.isEmpty() || CConfig == ConfigName))
226 {
227 if (CVersion.isEmpty() || CVersion > String(CurrentVersion) ||
228 (DowngradesAllowed && CVersion != String(CurrentVersion)))
229
230
231 return Action == DONT_DO_UPDATE ? UPDATE_AVAILABLE : DoOTAUpdate(firmwareURL, Action);
232 foundProfile = true;
233 }
234 }
235 return foundProfile ? NO_UPDATE_AVAILABLE : NO_UPDATE_PROFILE_FOUND;
236 }
