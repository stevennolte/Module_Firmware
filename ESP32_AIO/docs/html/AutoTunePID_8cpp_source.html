112{

148void AutoTunePID::performAutoTune(float currentInput)
149{
150 static unsigned long lastToggleTime = 0;
151 static bool outputState = true;
152 static int oscillationCount = 0;
153 static unsigned long oscillationStartTime = 0;
154
155 unsigned long currentTime = millis();
156
157 // Determine the output range based on the oscillation mode
158 float highOutput, lowOutput;
159 switch (_oscillationMode) {
160 case OscillationMode::Normal:
161 highOutput = _maxOutput;
162 lowOutput = _minOutput;
163 break;
164 case OscillationMode::Half:
165 highOutput = (_maxOutput + _minOutput) / 2.0f + (_maxOutput - _minOutput) / 4.0f; // 3/4 of the range
166 lowOutput = (_maxOutput + _minOutput) / 2.0f - (_maxOutput - _minOutput) / 4.0f; // 1/4 of the range
167 break;
168 case OscillationMode::Mild:
169 highOutput = (_maxOutput + _minOutput) / 2.0f + (_maxOutput - _minOutput) / 8.0f; // 5/8 of the range
170 lowOutput = (_maxOutput + _minOutput) / 2.0f - (_maxOutput - _minOutput) / 8.0f; // 3/8 of the range
171 break;
172 }
173
174 // Toggle output every second to induce oscillations
175 if (currentTime - lastToggleTime >= 1000) {
176 outputState = !outputState;
177 _output = outputState ? highOutput : lowOutput;
178 lastToggleTime = currentTime;
179
180 if (oscillationCount == 0) {
181 oscillationStartTime = currentTime;
182 }
183 oscillationCount++;
184
185 // After the specified number of oscillations, calculate Ku and Tu
186 if (oscillationCount >= _oscillationSteps) {
187 _oscillationPeriod = (currentTime - oscillationStartTime) / (float)(_oscillationSteps * 1000); // Period in seconds
188 _ultimateGain = (4.0f * (highOutput - lowOutput)) / (PI * (highOutput - lowOutput)); // Simplified amplitude
189
190 // Estimate T and L from the system response
191 _processTimeConstant = _oscillationPeriod / 2.0f; // Approximate T as half the oscillation period
192 _deadTime = _oscillationPeriod / 4.0f; // Approximate L as a quarter of the oscillation period
193
194 // Calculate Ti and Td based on T and L
195 _integralTime = 2.0f * _deadTime;
196 _derivativeTime = _deadTime / 2.0f;
197
198 // Calculate PID gains based on the selected tuning method
199 switch (_method) {
200 case TuningMethod::ZieglerNichols:
201 calculateZieglerNicholsGains();
202 break;
203 case TuningMethod::CohenCoon:
204 calculateCohenCoonGains();
205 break;
206 case TuningMethod::IMC:
207 calculateIMCGains();
208 break;
209 case TuningMethod::TyreusLuyben:
210 calculateTyreusLuybenGains();
211 break;
212 case TuningMethod::LambdaTuning:
213 calculateLambdaTuningGains();
214 break;
215 default:
216 break;
217 }
218
219 // Reset oscillation count and return to normal operation
220 oscillationCount = 0;
221 _operationalMode = OperationalMode::Normal; // Switch back to Normal mode after tuning
222 }
223 }
224}

255void AutoTunePID::calculateLambdaTuningGains()
256{
257 // Ensure lambda is set and valid
258 if (_lambda <= 0) {
259 _lambda = 0.5f * _processTimeConstant; // Default value for lambda
260 }
261
262 // Calculate Kp, Ki, and Kd using the Lambda Tuning (CLD) formula
263 _kp = _processTimeConstant / (_ultimateGain * (_lambda + _deadTime));
264 _ki = _kp / _processTimeConstant;
265 _kd = _kp * 0.5f * _deadTime;

268void AutoTunePID::computePID()
269{
270 // Calculate error
271 _error = _setpoint - _input;
272
273 // If error is very small, treat it as zero
274 if (abs(_error) < 0.001) {
275 _error = 0;
276 }
277
278 // Calculate PID terms
279 float P = _kp * _error;
280 float I = _ki * _integral;
281 float D = _kd * _derivative;
282
283 // Calculate output
284 _output = P + I + D;
285 _output = constrain(_output, _minOutput, _maxOutput);
286}
